from __future__ import annotations

import json
import os
from pathlib import Path
from typing import Any

from ascend_eitp.contracts.execution import production_execution_contract


class GroundingDinoAdapter:
    """Original IDEA-Research GroundingDINO entrypoint (build_model + raw .pth).

    Loads the repo-owned model (SwinT backbone + BERT text encoder) straight
    from a config .py and a .pth checkpoint, then runs forward on the device the
    pipeline passes in. Deformable attention falls back to its pure-torch path
    automatically because inputs are not CUDA tensors, so the same code runs on
    CPU baseline and NPU candidate.
    """

    model_id = "grounding-dino"

    def __init__(self, config: dict[str, Any] | None = None):
        self.config = config or {}

    def load_model(self, device: str):
        self._ensure_device_import(device)
        import torch

        from groundingdino.models import build_model
        from groundingdino.util.misc import clean_state_dict
        from groundingdino.util.slconfig import SLConfig

        config_path = self._repo_path(self.config.get("config_path", ""))
        if not config_path.is_file():
            raise FileNotFoundError(f"GroundingDINO model config not found: {config_path}")
        args = SLConfig.fromfile(str(config_path))
        args.device = device
        model = build_model(args)

        checkpoint_path = self._pick_checkpoint()
        if checkpoint_path is None:
            candidates = self.config.get("checkpoint_candidates") or []
            base = self._repo_root()
            raise FileNotFoundError(
                "no GroundingDINO checkpoint found among candidates "
                f"{[str(c) for c in candidates]} under repo {base}"
            )
        checkpoint = torch.load(str(checkpoint_path), map_location="cpu", weights_only=False)
        state = checkpoint.get("model", checkpoint)
        model.load_state_dict(clean_state_dict(state), strict=False)
        model.to(device).eval()
        return {"model": model, "device": device, "checkpoint": str(checkpoint_path)}

    def load_inputs(self, input_path: str | None, device: str):
        import json as _json

        path = Path(input_path or self.config.get("input_json", "input.json"))
        payload = _json.loads(Path(path).read_text(encoding="utf-8"))
        image_path = Path(payload["image_path"])
        if not image_path.is_absolute():
            image_path = Path(path).parent / image_path
        image_path = image_path.resolve()

        import PIL.Image as PILImage

        return {
            "image_path": str(image_path),
            "image_pil": PILImage.open(image_path).convert("RGB"),
            "text": payload["text"],
            "box_threshold": float(payload.get("box_threshold", self.config.get("box_threshold", 0.25))),
            "text_threshold": float(payload.get("text_threshold", self.config.get("text_threshold", 0.25))),
        }

    def run_inference(
        self, model: dict[str, Any], inputs: dict[str, Any], device: str
    ) -> dict[str, Any]:
        import torch

        torch_model = model["model"]
        image = self._preprocess(inputs["image_pil"]).to(device)
        caption = self._preprocess_caption(inputs["text"])

        with torch.inference_mode():
            outputs = torch_model(image[None], captions=[caption])

        logits = outputs["pred_logits"].cpu().sigmoid()[0]
        boxes_norm = outputs["pred_boxes"].cpu()[0]
        keep = logits.max(dim=1)[0] > inputs["box_threshold"]
        logits = logits[keep]
        boxes_norm = boxes_norm[keep]

        scores = logits.max(dim=1)[0]
        phrases = self._decode_phrases(
            torch_model, logits, caption, inputs["text_threshold"]
        )

        width, height = inputs["image_pil"].size
        boxes = [
            self._cxcywh_to_xyxy_px(box, width, height)
            for box in boxes_norm.tolist()
        ]
        return {
            "boxes": boxes,
            "scores": [float(score) for score in scores.tolist()],
            "labels": [str(phrase) for phrase in phrases],
            "detection_count": len(boxes),
            "score_threshold": inputs["box_threshold"],
            "image_path": inputs["image_path"],
            "text": inputs["text"],
            "execution_contract": production_execution_contract(
                output_evidence="final_task_predictions",
                input_provenance=inputs["image_path"],
                checkpoint_provenance=str(model["checkpoint"]),
                stages=["image_preprocess", "grounding_forward", "detection_postprocess"],
            ),
        }

    # ---- helpers ---------------------------------------------------------

    @staticmethod
    def _ensure_device_import(device: str) -> None:
        if str(device).startswith("npu") or str(device) == "ascend":
            import torch_npu  # noqa: F401  (registers the npu backend)

    def _repo_root(self) -> Path:
        base = Path(
            os.environ.get("EITP_MODEL_REPO_ROOT")
            or self.config.get("repo_root")
            or "."
        ).expanduser()
        return base.resolve()

    def _repo_path(self, value: str) -> Path:
        path = Path(value).expanduser()
        if not path.is_absolute():
            path = self._repo_root() / path
        return path.resolve()

    def _pick_checkpoint(self) -> Path | None:
        candidates = self.config.get("checkpoint_candidates") or []
        for candidate in candidates:
            text = str(candidate or "").strip()
            if not text:
                continue
            path = self._repo_path(text)
            if path.is_file():
                return path
        return None

    @staticmethod
    def _preprocess_caption(caption: str) -> str:
        result = caption.lower().strip()
        return result if result.endswith(".") else result + "."

    @staticmethod
    def _preprocess(image_pil):
        import groundingdino.datasets.transforms as T

        transform = T.Compose(
            [
                T.RandomResize([800], max_size=1333),
                T.ToTensor(),
                T.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225]),
            ]
        )
        image, _ = transform(image_pil, None)
        return image

    @staticmethod
    def _decode_phrases(
        torch_model: Any, logits: Any, caption: str, text_threshold: float
    ) -> list[str]:
        from groundingdino.util.utils import get_phrases_from_posmap

        tokenizer = torch_model.tokenizer
        tokenized = tokenizer(caption)
        phrases = []
        for logit in logits:
            posmap = (logit > text_threshold).bool()
            phrase = get_phrases_from_posmap(posmap, tokenized, tokenizer).replace(".", "")
            phrases.append(phrase)
        return phrases

    @staticmethod
    def _cxcywh_to_xyxy_px(box: list[float], width: int, height: int) -> list[float]:
        cx, cy, bw, bh = (float(value) for value in box)
        return [
            (cx - bw / 2.0) * width,
            (cy - bh / 2.0) * height,
            (cx + bw / 2.0) * width,
            (cy + bh / 2.0) * height,
        ]
